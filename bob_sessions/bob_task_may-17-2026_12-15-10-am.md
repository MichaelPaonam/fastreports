    """Install pytest if not available."""

    """Run unit tests."""
    print("\n" + "="*70)

    """Run integration tests."""

    print("RUNNING INTEGRATION TESTS")

    """Run all tests."""
    print("\n" + "="*70)

        [sys.executable, "-m", "pytest", "tests/", "-v"],

    )
    return result.returncode

    """Main test runner."""
    print("FastReports Test Suite")
    print("="*70)
    
    # Check pytest
    if not check_pytest_installed():
        print("pytest not found. Installing...")
        try:

            print(f"Error installing pytest: {e}")
            print("Please install pytest manually: pip install pytest")
            return 1
    
    # Parse command line arguments
    if len(sys.argv) > 1:
        test_type = sys.argv[1].lower()
        if test_type == "unit":
            return run_unit_tests()
        elif test_type == "integration":
            return run_integration_tests()
        elif test_type == "all":
            return run_all_tests()
        else:
            print(f"Unknown test type: {test_type}")
            print("Usage: python run_tests.py [unit|integration|all]")
            return 1
    else:
        # Run all tests by default
        return run_all_tests()

### Test Naming Convention

- Test files: `test_<module_name>.py`

def test_example_functionality(sample_dataframe):
    """Test description."""
    # Arrange

@pytest.mark.unit

    """Integration test."""

    """Slow test."""
    pass

   - Ensure `src/` is in Python path
   - Check that `__init__.py` files exist

2. **Fixture Not Found**
   - Verify fixture is defined in `conftest.py`
   - Check fixture scope (function, class, module, session)

3. **Test Failures**
   - Check test data and expectations
   - Verify module implementations match test assumptions
   - Review error messages and stack traces

1. **Test Independence**: Each test should be independent and not rely on others

5. **Test Edge Cases**: Include tests for boundary conditions
6. **Keep Tests Fast**: Unit tests should run quickly
7. **Meaningful Names**: Use descriptive test names
8. **Documentation**: Add docstrings to test functions
